// 날짜는 "일 번호"(1970-01-01부터 센 날 수)로 다룬다. 한국서버 시간대는 UTC+9.
const DAY = 86_400;
const UTC9 = 9 * 3600;

/** 유닉스 타임스탬프(초) → UTC+9 기준 일 번호 */
export function dayFromTimestamp(ts: number): number {
  return Math.floor((ts + UTC9) / DAY);
}

/** 지금 이 순간의 한국 날짜 일 번호 */
export function todayKr(now = Date.now()): number {
  return dayFromTimestamp(Math.floor(now / 1000));
}

/**
 * 일본서버가 한국서버보다 앞선 날 수. 한국서버 일정 데이터가 없는 운영 선물(일본 공지)을
 * 한국서버 예상일로 옮길 때만 쓴다
 */
export const SERVER_GAP_DAYS = 476;

function parts(day: number) {
  const d = new Date(day * DAY * 1000);
  return { y: d.getUTCFullYear(), m: d.getUTCMonth() + 1, d: d.getUTCDate(), w: d.getUTCDay() };
}

const WEEKDAYS = ["일", "월", "화", "수", "목", "금", "토"];

/** 11/19 */
export function formatMd(day: number): string {
  const p = parts(day);
  return `${p.m}/${p.d}`;
}

/** 2026년 11월 23일 (월) */
export function formatLong(day: number): string {
  const p = parts(day);
  return `${p.y}년 ${p.m}월 ${p.d}일 (${WEEKDAYS[p.w]})`;
}

/** 2026.11.23 */
export function formatFull(day: number): string {
  const p = parts(day);
  return `${p.y}.${String(p.m).padStart(2, "0")}.${String(p.d).padStart(2, "0")}`;
}

/** 11/19 12:00 (타임스탬프 기준, UTC+9) */
export function formatMdTime(ts: number): string {
  const d = new Date((ts + UTC9) * 1000);
  return `${d.getUTCMonth() + 1}/${d.getUTCDate()} ${String(d.getUTCHours()).padStart(2, "0")}:${String(d.getUTCMinutes()).padStart(2, "0")}`;
}

/** 그 달 1일의 일 번호 (month는 1–12, 넘치면 다음 해로) */
export function monthStart(year: number, month: number): number {
  return Math.floor(Date.UTC(year, month - 1, 1) / (DAY * 1000));
}

/** 일 번호가 속한 해와 달 */
export function monthOf(day: number): { y: number; m: number } {
  const p = parts(day);
  return { y: p.y, m: p.m };
}

export function dayOfMonth(day: number): number {
  return parts(day).d;
}

/** 0 = 일요일 */
export function weekdayIndex(day: number): number {
  return parts(day).w;
}

export function weekday(day: number): string {
  return WEEKDAYS[parts(day).w];
}

/** 종료까지 이 일수 이하로 남으면 강조색으로 표시 */
export const URGENT_DAYS = 3;

/**
 * 진행 상태와 상대 시간 문구.
 * 진행 중이면 "n일 남음"(마지막 날이면 "오늘 종료"), 시작 전이면 "n일 후 시작"(내일이면 "내일 시작"), 끝났으면 "종료"
 */
export function relative(
  startDay: number,
  endDay: number,
  today: number,
): { state: "live" | "soon" | "ended"; text: string; urgent: boolean } {
  if (today < startDay) {
    const n = startDay - today;
    return { state: "soon", text: n === 1 ? "내일 시작" : `${n}일 후 시작`, urgent: false };
  }
  if (today <= endDay) {
    const n = endDay - today;
    return { state: "live", text: n === 0 ? "오늘 종료" : `${n}일 남음`, urgent: n <= URGENT_DAYS };
  }
  return { state: "ended", text: "종료", urgent: false };
}

/** 생일·날짜까지 남은 날: 오늘 / 내일 / n일 후 */
export function untilText(n: number): string {
  return n === 0 ? "오늘" : n === 1 ? "내일" : `${n}일 후`;
}

/** D-7, D-Day */
export function formatDday(diff: number): string {
  return diff === 0 ? "D-Day" : diff > 0 ? `D-${diff}` : `D+${-diff}`;
}

const BASE = "https://gametora.com/data";
const REVALIDATE = 3600;

type Manifest = Record<string, string>;

async function manifest(): Promise<Manifest> {
  return fetchJson<Manifest>(`${BASE}/manifests/umamusume.json`, REVALIDATE);
}

/** 현재 해시가 붙은 데이터 파일 주소 */
export async function dataUrl(key: string): Promise<string> {
  const m = await manifest();
  const hash = m[key];
  if (!hash) throw new Error(`manifest에 ${key} 없음`);
  return `${BASE}/umamusume/${key}.${hash}.json`;
}

export async function load<T>(key: string): Promise<T> {
  return fetchJson<T>(await dataUrl(key), REVALIDATE);
}

/** 실패하면 fallback을 돌려주고 서버 로그에 남긴다 */
async function safe<T>(what: string, fn: () => Promise<T>, fallback: T): Promise<T> {
  try {
    return await fn();
  } catch (err) {
    logFailure(what, err);
    return fallback;
  }
}

export function cardIndex(cards: Card[]) {
  const train = new Map<number, Card>();
  const support = new Map<number, Card>();
  for (const c of cards) (c.lane === "train" ? train : support).set(c.id, c);
  return { train, support, get: (lane: Lane, id: number) => (lane === "train" ? train : support).get(id) };
}

/** 시작 전 배너는 이미지가 아직 없을 수 있어서 있는지 확인한다 (하루 캐시) */
async function existing(url: string): Promise<string | null> {
  try {
    const res = await fetch(url, { method: "HEAD", next: { revalidate: 86_400 } });
    return res.ok ? url : null;
  } catch {
    return null;
  }
}

/** 날짜가 정해지지 않은 자리표시 배너(2099년 등)를 거른다 */
const FAR_FUTURE = Date.UTC(2035, 0, 1) / 1000;

export async function getBanners(cards?: Card[]): Promise<Banner[]> {
  const list = cards ?? (await getCards());
  const idx = cardIndex(list);
  return safe(
    "gametora 가챠 배너",
    async () => {
      const [charStd, supStd, special, timeline] = await Promise.all([
        load<RawBanner[]>("ko/gacha/char-standard"),
        load<RawBanner[]>("ko/gacha/support-standard"),
        safe("gametora 특수 가챠", () => load<RawBanner[]>("ko/gacha/special"), []),
        safe("gametora 미래 일정", () => load<RawTimeline>("ko/foresight/timeline"), {
          future_char_banners: [],
          future_support_banners: [],
        }),
      ]);
      const estimated = new Set(
        [...timeline.future_char_banners, ...timeline.future_support_banners].filter((b) => b.is_estimated).map((b) => b.id),
      );
      const toBanner = (b: RawBanner, lane: Lane, paidOnly: boolean): Banner => ({
        id: b.id,
        lane,
        startTs: b.start,
        endTs: b.end,
        start: dayFromTimestamp(b.start),
        end: dayFromTimestamp(b.end),
        pickups: (b.pickups ?? []).flatMap(([id, rate, isNew]) => {
          const card = idx.get(lane, id);
          return card ? [{ card, rate, isNew }] : [];
        }),
        lineup: b.lineup,
        estimated: estimated.has(b.id),
        paidOnly,
        star3Up: !!b.rarityOdds && b.rarityOdds[0] > 30_000,
        art: bannerArt(b.id),
      });
      // 유료 전용 특수 가챠만 일정에 넣는다. 나머지 특수 가챠는 정체를 알 수 없는 상설·월간 배너다
      const paid = special
        // 몇 년씩 열려 있는 상설 유료 가챠는 일정이 아니라서 뺀다
        .filter((b) => b.restriction === "premium" && b.end - b.start < 60 * 86_400)
        .map((b) => {
          const ids = Object.values(b.lineup).flat();
          const lane: Lane = ids.some((id) => idx.support.has(id)) && !ids.some((id) => idx.train.has(id)) ? "support" : "train";
          const banner = toBanner(b, lane, true);
          // 유료 전용 배너는 pickups가 비어 있어서, 골라 뽑는 ★3·SSR 목록(카드마다 0.1% 이상)을 픽업으로 본다
          if (banner.pickups.length === 0) {
            banner.pickups = Object.entries(b.lineup)
              .filter(([rate]) => Number(rate) >= 1000)
              .flatMap(([rate, ids]) =>
                ids.flatMap((id) => {
                  const card = idx.get(lane, id);
                  return card && card.rarity === 3 ? [{ card, rate: Number(rate), isNew: false }] : [];
                }),
              );
          }
          return banner;
        });
      const all = [...charStd.map((b) => toBanner(b, "train", false)), ...supStd.map((b) => toBanner(b, "support", false)), ...paid]
        .filter((b) => b.startTs < FAR_FUTURE)
        .sort((a, b) => a.startTs - b.startTs || a.id - b.id);
      const now = Date.now() / 1000;
      await Promise.all(
        all.filter((b) => b.startTs > now && b.art).map(async (b) => {
          b.art = await existing(b.art as string);
        }),
      );
      return all;
    },
    [],
  );
}

  rarity: number;

  weight: number;
  pickup: boolean;

/** 배너의 lineup을 카드별 확률 목록으로. ★ 높은 순, 픽업 먼저 */
export function buildPool(banner: Banner, get: (lane: Lane, id: number) => Card | undefined): PoolCard[] {
  const pickupIds = new Set(banner.pickups.map((p) => p.card.id));
  const pool = Object.entries(banner.lineup).flatMap(([weight, ids]) =>
    ids.flatMap((id): PoolCard[] => {
      const c = get(banner.lane, id);
      return c ? [{ id, name: c.name, title: c.title, rarity: c.rarity, weight: Number(weight), pickup: pickupIds.has(id), art: c.art, supportType: c.supportType }] : [];
    }),
  );
  return pool.sort((a, b) => b.rarity - a.rarity || Number(b.pickup) - Number(a.pickup) || b.weight - a.weight || a.id - b.id);
}

export function formatRate(weight: number): string {

import type { PoolCard } from "./gachaPool";

export type Table = { cards: PoolCard[]; cum: number[]; total: number };

export function toTable(cards: PoolCard[], weightOf: (c: PoolCard) => number): Table {
  const cum: number[] = [];
  let total = 0;
  for (const c of cards) {
    total += weightOf(c);
    cum.push(total);
  }
  return { cards, cum, total };
}

/** r은 [0, 1) 난수. 테스트에서 고정값을 넣을 수 있게 받는다 */
export function draw(t: Table, r = Math.random()): PoolCard {
  const x = r * t.total;
  let lo = 0;
  let hi = t.cum.length - 1;
  while (lo < hi) {
    const mid = (lo + hi) >> 1;
    if (t.cum[mid] > x) hi = mid;
    else lo = mid + 1;
  }
  return t.cards[lo];
}

export type Odds = {
  /** 1–9번째 칸 */
  normal: Table;
  /** 10번째 칸 */
  last: Table;
  /** 레어도별 확률 합, 인덱스가 레어도 (0은 비움) */
  sums: number[];
  lastSums: number[];
};

// 10번째 칸: ★1 확률을 빼고 그만큼을 ★2 카드에 똑같이 나눠 더한다 (gametora 시뮬레이터와 같은 방식).
// ★2 카드가 없는 배너는 옮길 곳이 없어서 10번째 칸도 일반 확률 그대로다
export function buildOdds(pool: PoolCard[]): Odds {
  const sums = [0, 0, 0, 0];
  for (const c of pool) sums[c.rarity] += c.weight;
  const normal = toTable(pool, (c) => c.weight);
  const star2 = pool.filter((c) => c.rarity === 2).length;
  if (star2 === 0) return { normal, last: normal, sums, lastSums: sums };

  const bonus = sums[1] / star2;
  const last = toTable(pool, (c) => (c.rarity === 1 ? 0 : c.rarity === 2 ? c.weight + bonus : c.weight));
  return { normal, last, sums, lastSums: [0, 0, sums[2] + sums[1], sums[3]] };
}

  const eventFree = sum("free") - competitionFree - giftFree;

  const login = Math.floor(days / LOGIN_BONUS.days) * LOGIN_BONUS.jewels;
  const teamWeeks = input.teamClass === 0 ? 0 : mondaysBetween(input.today, input.target);
  const teamTrials = teamWeeks * TEAM_TRIALS_WEEKLY[input.teamClass];

  // 월정액과 같이, 지금 기간부터 30일마다 한 기간씩 보상을 끝까지 받는다고 본다
  const passPeriods = input.pass === "none" ? 0 : Math.ceil(days / TRAINING_PASS.days);
  const std = TRAINING_PASS.standard;
  const pre = input.pass === "premium" ? TRAINING_PASS.premium : { paidOnBuy: 0, free: 0, trainTickets: 0, supportTickets: 0 };
  const passFree = passPeriods * (std.free + pre.free);
  const passPaid = passPeriods * pre.paidOnBuy;
  const passTrainTickets = passPeriods * (std.trainTickets + pre.trainTickets);
  const passSupportTickets = passPeriods * (std.supportTickets + pre.supportTickets);

  return {
    days,
    daily,
    monthlyFree,
    monthlyPaid,
    passPeriods,
    passFree,
    passPaid,
    passTrainTickets,
    passSupportTickets,
    incomes,
    eventFree,
    competitionFree,
    login,
    teamWeeks,
    teamTrials,
    giftFree,
    free: input.free + daily + monthlyFree + passFree + eventFree + competitionFree + giftFree + login + teamTrials,
    paid: input.paid + monthlyPaid + passPaid,
    trainTickets: input.trainTickets + passTrainTickets + sum("trainTickets"),
    supportTickets: input.supportTickets + passSupportTickets + sum("supportTickets"),
  };
}

/** 목표 픽업에서 돌릴 수 있는 횟수. 유료 전용 가챠는 유료 쥬얼만 쓰고 티켓은 못 쓴다 */
export function pullsFor(p: Plan, lane: "train" | "support", paidOnly: boolean) {
  if (paidOnly) {
    const fromJewels = Math.floor(p.paid / JEWELS_PER_PULL);
    return { fromJewels, fromTickets: 0, total: fromJewels };
  }
  const fromJewels = Math.floor((p.free + p.paid) / JEWELS_PER_PULL);
  const fromTickets = lane === "train" ? p.trainTickets : p.supportTickets;
  return { fromJewels, fromTickets, total: fromJewels + fromTickets };
}

const KIND_NAME = { cm: "챔피언스 미팅", legend: "레전드 레이스", loh: "리그 오브 히어로즈" } as const;

/** 고른 단계로 대회 보상을 기간 보상 목록에 끼워 넣을 수 있는 모양으로 바꾼다. 0쥬얼인 대회는 뺀다 */
export function competitionIncomes(schedule: Scheduled[], c: CompetitionInput): Income[] {
  return schedule.flatMap((s): Income[] => {
    if (s.kind === "legend") {
      if (!c.legendRace) return [];
      const starts = s.raceStartsKr?.length ? s.raceStartsKr : [s.startKr];
      const base = s.name ? `${KIND_NAME.legend} ${s.name}` : KIND_NAME.legend;
      return starts.map((day, i) => ({
        id: `${s.id}#${i + 1}`,
        source: "legend",
        name: starts.length > 1 ? `${base} (${i + 1}/${starts.length})` : base,
        startKr: day,
        endKr: s.endKr,
        free: LEGEND_RACE_FIRST_WIN,
        trainTickets: 0,
        supportTickets: 0,
      }));
    }
    const free =
      s.kind === "cm" ? cmJewels(c.cmLeague, c.cmResult, s.startKr) : lohJewels(c.lohRank);
    if (free === 0) return [];
    // 챔피언스 미팅은 결승이 끝나야 보상이 나오므로 마지막 날을 기준으로 센다
    const day = s.kind === "cm" ? s.endKr : s.startKr;
    return [
      {
        id: s.id,
        source: s.kind,
        name: s.name ? `${KIND_NAME[s.kind]} ${s.name}` : KIND_NAME[s.kind],
        startKr: day,
        endKr: s.endKr,
        free,
        trainTickets: 0,
        supportTickets: 0,
      },
    ];
  });
}

/** (today, target) 사이의 월요일 수. 일 번호 0(1970-01-01)은 목요일이다 */
export function mondaysBetween(today: number, target: number): number {
  let n = 0;
  for (let d = today + 1; d < target; d++) if ((d + 3) % 7 === 0) n++;
  return n;
}

    setStore((prev) => {
      const next = { ...prev, [banner.id]: fn(prev[banner.id] ?? emptyRecord(lane)) };
      persist(next);
      return next;
    });

    const cards = Array.from({ length: times }, (_, i) => draw(times === 10 && i === 9 ? last : normal));
    const nextRound = round + 1;

      const counts: BannerRecord["counts"] = [...r.counts];
      const got = { ...r.got };

        got[c.id] = (got[c.id] ?? 0) + 1;

let cache: SearchItem[] | null = null;

export function SearchModal({ open, onClose }: { open: boolean; onClose: () => void }) {
  const router = useRouter();
  const [items, setItems] = useState<SearchItem[] | null>(cache);
  const [failed, setFailed] = useState(false);
  const [query, setQuery] = useState("");
  const [active, setActive] = useState(0);
  const inputRef = useRef<HTMLInputElement>(null);
  const dialogRef = useRef<HTMLDialogElement>(null);

  useEffect(() => {
    const d = dialogRef.current;
    if (!d) return;
    if (open && !d.open) {
      d.showModal();
      setQuery("");
      setActive(0);
      setTimeout(() => inputRef.current?.focus(), 0);
    } else if (!open && d.open) d.close();
  }, [open]);

  useEffect(() => {
    if (!open || items) return;
    fetch("/api/search")
      .then((r) => (r.ok ? r.json() : Promise.reject(r.status)))
      .then((data: SearchItem[]) => {
        cache = data;
        setItems(data);
      })
      .catch(() => setFailed(true));
  }, [open, items]);

  const results = useMemo(() => (items ? searchItems(items, query) : []), [items, query]);

  useEffect(() => setActive(0), [query]);

  const go = (it: SearchItem) => {
    onClose();
    router.push(it.href);
  };

  const onKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === "ArrowDown") {
      e.preventDefault();
      setActive((a) => Math.min(a + 1, results.length - 1));
    } else if (e.key === "ArrowUp") {
      e.preventDefault();
      setActive((a) => Math.max(a - 1, 0));
    } else if (e.key === "Enter" && results[active]) {
      e.preventDefault();
      go(results[active]);
    }
  };

  let index = -1;

  return (
    <dialog
      ref={dialogRef}

      onClose={onClose}

      }}

import type { SupportType } from "./gametora";

export type SearchGroup = "육성 우마무스메" | "서포트 카드" | "픽업" | "이벤트";

export type SearchItem = { group: SearchGroup; title: string; sub: string; href: string; img?: string; supportType?: SupportType };

export const SEARCH_GROUPS: SearchGroup[] = ["육성 우마무스메", "서포트 카드", "픽업", "이벤트"];

const norm = (s: string) => s.toLowerCase().replace(/[\s·・\-\[\]【】()]/g, "");

/** 이름 일부로 찾는다. 앞에서 맞는 것을 먼저, 묶음마다 최대 limit개 */
export function searchItems(items: SearchItem[], query: string, limit = 6): SearchItem[] {
  const q = norm(query);
  if (!q) return [];
  const scored = items.flatMap((it) => {
    const t = norm(it.title);
    const s = norm(it.sub);
    const at = t.indexOf(q);
    if (at >= 0) return [{ it, score: at === 0 ? 0 : 1 }];
    if (s.includes(q)) return [{ it, score: 2 }];
    return [];
  });
  scored.sort((a, b) => a.score - b.score);
  return SEARCH_GROUPS.flatMap((g) => scored.filter((x) => x.it.group === g).slice(0, limit).map((x) => x.it));
}

| 뽑기 시뮬레이터 | `/simulator` | 실제 배너와 제공 비율로 단차 / 10연차, 천장 포인트(200) 교환, 배너별 뽑기 기록과 통계 |
| 사료계산기 | `/calculator` | 지금 가진 쥬얼·티켓과 받을 보상을 넣고 목표 픽업을 고르면, 그날까지 모이는 쥬얼과 뽑기 횟수를 계산 |
| 생일달력 | `/birthdays` | 1월~6월 캐릭터 생일을 달별 달력으로, 다음 생일과 D-day |
| 이벤트 일정 | `/events` | 게임 이벤트를 한 달 달력에 기간 막대로 표시하고 아래에 목록으로. 경기 / 스토리 / 육성 / 기념 미션 필터 |

### 사료계산기 상세

- 입력: 무료·유료 쥬얼, 육성·서포트 티켓, 월정액, 시즌 패스(무료 / 프리미엄), 팀 경기장 클래스, 챔피언스 미팅 리그와 결승 결과, 리그 오브 히어로즈 최종 랭크, 레전드 레이스 1착 여부, 목표 픽업
- 목표로 고를 수 있는 픽업: 오늘부터 240일 안에 시작하는 픽업
- 더하는 보상: 로그인 보너스, 기간 한정 미션, 스토리 이벤트 보상, 경기 보상, 선물 등. 각 수치의 출처는 `lib/jewelSources.ts`에 적혀 있음
- 유료 쥬얼 전용 가챠는 유료 쥬얼로만 계산
- 신데렐라 그레이 방영 기념 쥬얼 캠페인은 한일 동시 진행이라 계산에서 제외

### 이벤트 일정에 넣는 이벤트

레전드 레이스, 챔피언스 미팅, 리그 오브 히어로즈, 스토리 이벤트, 기능시험, 레이싱 카니발, 팀 스카우트, 주년 기념, G1 기념, 기념 미션. 마스터즈 챌린지는 개최 날짜 자료를 찾지 못해 넣지 않았습니다.

## 규칙

- 모든 데이터는 일본서버 기준입니다.
- 한국서버 예상일 = 일본서버 날짜 + 476일(고정). 화면에는 한국서버 예상일을 메인으로, 일본서버 날짜를 보조로 함께 표시합니다.
- 캐릭터·카드 이름은 한국서버 공식 한글명을 우선하고, 없으면 일본어 원문으로 표시합니다.
- 사료계산기 입력값과 뽑기 기록은 사용하는 브라우저의 `localStorage`에만 저장되며 기기 간 동기화는 없습니다.

## 데이터 출처

외부 API를 실시간으로 호출하고, 응답은 Next.js 데이터 캐시에 둡니다.

- [umapyoi.net](https://api.umapyoi.net/): 픽업 일정, 일러스트, 생일
- [gametora.com](https://gametora.com/ko/umamusume): 배너·제공 비율, 한국어 이름, 미션·이벤트 보상, 이벤트 일정
- [uma.guide](https://uma.guide/): 게임 데이터·보상 확인용 참고 자료
- 고정 보상표(로그인 보너스, 팀 경기장, 챔피언스 미팅 등)는 공략 사이트·공식 공지를 근거로 `lib/jewelSources.ts`에 직접 정리

# Product

<!-- impeccable:product-schema 1 -->

## Platform

web

## Stack

Next.js (App Router), Vercel 배포. 로그인·서버 DB 없음, 사용자 입력은 브라우저 저장소. 개발 서버는 localhost:5640. (README에서 확정, 2026-11-03 로그인 제거)

## Users

우마무스메 한국서버 플레이어 전체를 대상으로 하는 공개 커뮤니티 서비스. 처음 들어온 사람도 설명 없이 이해해야 한다. 주로 "다음 픽업까지 쥬얼을 얼마나 모을 수 있나", "언제 누가 나오나"를 판단하려는 상황에서 쓴다.

## Product Purpose

일본서버가 한국서버보다 476일 앞서 있다는 사실을 이용해, 한국서버 유저가 미래 픽업을 미리 보고 쥬얼(사료)을 계획하게 돕는다. 성공은 유저가 목표 픽업을 정하고 그때까지 모일 쥬얼을 숫자로 확인하는 것.

## Positioning

일본서버 데이터를 한국서버 예상일(+476일)로 환산해 메인 날짜로 보여주고, 그 날짜까지의 사료 계산을 한 서비스 안에서 이어준다.

## Operating Context

- 모든 데이터는 일본서버 기준이며 한국서버 예상일 = 일본서버 날짜 + 476일(고정). 한국 예상일이 메인, 일본 날짜는 보조로 함께 표시.
- 데이터 출처: api.umapyoi.net(가챠·일러·생일), gametora.com(사료량), uma.guide(보상·미션 보완). 실시간 호출.
- 로그인 없음. 누구나 모든 기능을 쓴다. 사료계산기 입력값과 뽑기 기록은 그 브라우저에만 저장되고 기기 간 동기화는 없다.

## Capabilities and Constraints

- 가챠: 과거·미래 픽업 일정표(타임라인/목록), 뽑기 시뮬레이터(단차/10연차, 천장 포인트, 기록·통계, 육성 우마무스메/서포트 카드 구분), 유료 쥬얼 전용 가챠 포함.
- 사료계산기: 보유 쥬얼, 보유 티켓(우마무스메/서포트), 월정액 여부, 목표 픽업 입력 → 목표 픽업까지 모이는 쥬얼 계산. 무료/유료 쥬얼 구분.
- 신데렐라 그레이 방영 기념 쥬얼 캠페인은 한일 동시 진행이므로 사료 계산에서 제외.
- 생일달력: 1월~6월 생일자만 표시(고정 범위).
- 이벤트 일정: 게임 내 이벤트 일정을 한국서버 예상일 기준으로 표시. gametora와 uma.guide 데이터 참고. (2026-11-03 패키지 기능을 대체)
- 이름 표기: 캐릭터명·카드 제목은 gametora 데이터의 한국어 표기(name_ko, title_ko) 우선, 없으면 일본어 원문.
- 메인 화면: 다음/진행 중 픽업과 그때까지의 사료 요약을 가장 먼저 보여주고, 각 기능으로 진입.

## Brand Commitments

서비스명 "우마노트". 로고·브랜드 자산 없음. 캐릭터·카드 일러스트는 umapyoi API 이미지를 사용해도 된다(사용자 확인).

## Evidence on Hand

외부 API 데이터 외에 자체 수치·후기·사용자 수 없음. 픽업 일정, 쥬얼 수치, 이용자 통계를 지어내지 않는다.

## Product Principles

1. 한국서버 예상일이 언제나 주인공이고, 일본서버 날짜는 근거로 곁에 둔다.
2. 숫자는 출처가 있는 것만 보여준다. 계산 결과에는 무엇을 넣고 뺐는지 드러낸다.
3. 누구나 가입·로그인 없이 바로 모든 기능을 쓴다.
4. 처음 온 사람도 첫 화면에서 "다음 픽업과 내 쥬얼"을 바로 이해한다.
